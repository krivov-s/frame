public class MyScriptDataSource : IScriptCommandDataSource
{
    public async Task<Result<List<dynamic>>> ExecuteAsync(AppCore appCore, ParamList? paramList = null, List<dynamic>? entities = null, IObjectStorage? storage = null)
    {
        List<dynamic> listMeta = EntityMetadataDataSourceService.GetEntityMetadataList(paramList);
        string msg = $"Кол-во записей в списке метаданных: {listMeta.Count}";
        appCore.Logger.LogInformation(msg);
        return Result<List<dynamic>>.SuccessWithMessage(listMeta, msg);
    }
}
